from contextlib import asynccontextmanager

from fastapi import FastAPI, Request
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse
from psycopg.errors import UniqueViolation
from sqlalchemy.exc import IntegrityError

from app.core.database import engine
from app.modules.config.router import router as config_router
from app.modules.tickets.import_csv import router as import_router
from app.modules.tickets.router import router as tickets_router
from app.modules.users.router import router as users_router


@asynccontextmanager
async def lifespan(app: FastAPI):
    try:
        yield
    finally:
        await engine.dispose()


app = FastAPI(title="售后工单智能辅助系统", lifespan=lifespan)
app.include_router(users_router)
app.include_router(config_router)
app.include_router(import_router)
app.include_router(tickets_router)


@app.exception_handler(RequestValidationError)
async def invalid_request(request: Request, error: RequestValidationError):
    # 前端只需要字段位置与错误原因；原始输入可能含登录密码。
    details = [
        {"type": item["type"], "loc": item["loc"], "msg": item["msg"]}
        for item in error.errors()
    ]
    return JSONResponse(status_code=422, content={"detail": details})


@app.exception_handler(IntegrityError)
async def duplicate_record(request: Request, error: IntegrityError):
    if isinstance(error.orig, UniqueViolation):
        return JSONResponse(status_code=409, content={"detail": "名称或用户名已存在"})
    raise error

    return {"status": "ok"}

**它做什么：** 先读文件、检查每一行，再把全部工单一次性写进数据库。任何一行不合格，本批都不写。

**固定写法：** UploadFile、APIRouter、model_validator、select、commit 是框架接口；AdminUser 复用已有管理员权限检查。

**能改什么、要一起改什么：** 4 MiB、5000 条是本阶段限制；英文列名和顺序是导入契约，前后端模板必须一起改。pending 必须无方案、resolved 必须有方案是业务规则。

```python
import csv
import io
from typing import Annotated, Literal, Self

from fastapi import APIRouter, HTTPException, UploadFile
from psycopg.errors import UniqueViolation
from pydantic import (
    BaseModel,
    ConfigDict,
    EmailStr,
    StringConstraints,
    ValidationError,
    model_validator,
)
from sqlalchemy import select
from sqlalchemy.exc import IntegrityError

from app.core.security import AdminUser, DbSession
from app.models import Product, ProductModule, Ticket, TicketCategory
from app.modules.tickets.schemas import CustomerName, Description, Priority, Title
from app.modules.tickets.service import add_record, validate_catalog

router = APIRouter(prefix="/api/tickets", tags=["CSV 导入"])
MAX_BYTES = 4 * 1024 * 1024
MAX_ROWS = 5000
Name = Annotated[str, StringConstraints(strip_whitespace=True, min_length=1, max_length=100)]


class ImportRow(BaseModel):
    model_config = ConfigDict(extra="forbid", hide_input_in_errors=True)
    import_key: Name
    title: Title
    description: Description
    customer_name: CustomerName
    customer_email: EmailStr
    product: Name
    module: Name
    category: Name
    priority: Priority
    status: Literal["pending", "resolved"]
    solution: Annotated[str, StringConstraints(strip_whitespace=True, max_length=10000)]

    @model_validator(mode="after")
    def check_solution(self) -> Self:
        if self.status == "resolved" and not self.solution:
            raise ValueError("已解决工单必须填写解决方案")
        if self.status == "pending" and self.solution:
            raise ValueError("待处理工单的解决方案应为空")
        return self


class ImportResult(BaseModel):
    imported: int
    ticket_ids: list[int]


def parse_csv(content: bytes) -> list[ImportRow]:
    try:
        text = content.decode("utf-8-sig")
    except UnicodeDecodeError as error:
        raise HTTPException(422, "CSV 必须保存为 UTF-8 编码") from error
    reader = csv.DictReader(io.StringIO(text, newline=""), strict=True)
    rows = []
    keys = set()
    try:
        if reader.fieldnames != list(ImportRow.model_fields):
            raise HTTPException(422, "CSV 表头与模板不一致，请保留字段顺序和英文名称")
        for number, values in enumerate(reader, start=1):
            if number > MAX_ROWS:
                raise HTTPException(422, f"每次最多导入 {MAX_ROWS} 条记录")
            if None in values or None in values.values():
                raise HTTPException(422, f"第 {number} 条记录的列数与表头不一致")
            try:
                row = ImportRow.model_validate(values)
            except ValidationError as error:
                reasons = "；".join(
                    f"{'.'.join(map(str, item['loc'])) or '记录'}: {item['msg']}"
                    for item in error.errors()
                )
                raise HTTPException(422, f"第 {number} 条记录：{reasons}") from error
            if row.import_key in keys:
                raise HTTPException(409, f"第 {number} 条记录的 import_key 在文件内重复")
            keys.add(row.import_key)
            rows.append(row)
    except csv.Error as error:
        raise HTTPException(422, f"CSV 第 {reader.line_num} 行格式错误：{error}") from error
    if not rows:
        raise HTTPException(422, "CSV 没有数据记录")
    return rows


@router.post("/import", response_model=ImportResult, status_code=201)
async def import_tickets(file: UploadFile, db: DbSession, user: AdminUser):
    # UploadFile 由 FastAPI 关闭；有界读取避免把任意大小的文件全部装入内存。
    content = await file.read(MAX_BYTES + 1)
    if len(content) > MAX_BYTES:
        raise HTTPException(413, "CSV 不能超过 4 MiB")
    rows = parse_csv(content)
    existing = set((await db.scalars(select(Ticket.import_key).where(
        Ticket.import_key.in_([row.import_key for row in rows])
    ))))
    products = {item.name: item for item in (await db.scalars(select(Product)))}
    modules = {(item.product_id, item.name): item for item in (await db.scalars(select(ProductModule)))}
    categories = {item.name: item for item in (await db.scalars(select(TicketCategory)))}
    tickets = []
    for number, row in enumerate(rows, start=1):
        if row.import_key in existing:
            raise HTTPException(409, f"第 {number} 条记录已经导入，请勿重复上传")
        product = products.get(row.product)
        category = categories.get(row.category)
        module = modules.get((product.id, row.module)) if product is not None else None
        if product is None or category is None or module is None:
            raise HTTPException(422, f"第 {number} 条记录：产品、所属模块或分类名称不存在")
        try:
            await validate_catalog(db, product.id, module.id, category.id)
        except HTTPException as error:
            raise HTTPException(error.status_code, f"第 {number} 条记录：{error.detail}") from error
        ticket = Ticket(
            **row.model_dump(exclude={"product", "module", "category"}),
            product_id=product.id, module_id=module.id, category_id=category.id, owner_id=user.id,
        )
        add_record(ticket, user.id, "imported", "管理员导入历史记录；时间为导入时间，非原始处理时间")
        tickets.append(ticket)
    db.add_all(tickets)
    try:
        await db.commit()
    except IntegrityError as error:
        await db.rollback()
        if isinstance(error.orig, UniqueViolation) and error.orig.diag.constraint_name == "tickets_import_key_key":
            raise HTTPException(409, "这些编号已被另一个请求导入，本批未写入") from error
        raise
    return ImportResult(imported=len(tickets), ticket_ids=[ticket.id for ticket in tickets])
```

import { z } from 'zod'

const errorSchema = z.object({
  detail: z.union([
    z.string(),
    z.array(z.object({
      loc: z.array(z.union([z.string(), z.number()])),
      msg: z.string(),
    })),
  ]),
})

export class ApiError extends Error {
  constructor(public readonly status: number, message: string) {
    super(message)
    this.name = 'ApiError'
  }
}

interface RequestOptions {
  token?: string
  method?: 'GET' | 'POST' | 'PUT' | 'PATCH'
  body?: unknown
  timeoutMs?: number
}

// 登录、配置、工单都需要 JSON 请求、错误展示和超时，因此在此复用。
export async function request<T>(
  path: string, schema: z.ZodType<T>, options: RequestOptions = {},
): Promise<T> {
  const headers = new Headers()
  if (options.token) headers.set('Authorization', `Bearer ${options.token}`)
  const isUpload = options.body instanceof FormData
  if (options.body !== undefined && !isUpload) headers.set('Content-Type', 'application/json')
  let body: BodyInit | undefined
  if (options.body instanceof FormData) body = options.body
  else if (options.body !== undefined) body = JSON.stringify(options.body)

  const response = await fetch('/api' + path, {
    method: options.method ?? 'GET',
    headers,
    body,
    signal: AbortSignal.timeout(options.timeoutMs ?? 10000),
  })

  if (!response.ok) {
    let message = `请求失败：HTTP ${response.status}`
    // FastAPI 的业务/校验错误是 JSON；未处理的服务端异常可能只有文本。
    if (response.headers.get('content-type')?.includes('application/json')) {
      const { detail } = errorSchema.parse(await response.json())
      message = typeof detail === 'string'
        ? detail
        : detail.map(issue => `${issue.loc.join('.')}: ${issue.msg}`).join('；')
    }
    throw new ApiError(response.status, message)
  }

  return schema.parse(await response.json())
}
```

import asyncio
import json
from datetime import UTC, datetime
from pathlib import Path

from pymilvus import AnnSearchRequest, DataType, Function, FunctionType, MilvusClient, RRFRanker

COLLECTION = "ticket_cases"


class IndexNotReady(RuntimeError):
    """缺少完整索引，或当前配置与构建时不一致。"""


class CaseIndex:
    def __init__(self, path: Path, model: str, base_url: str, dimensions: int):
        path.parent.mkdir(parents=True, exist_ok=True)
        self.metadata_path = Path(str(path) + ".index.json")
        self.signature = {"model": model, "base_url": base_url.rstrip("/"), "dimensions": dimensions}
        self.dimensions = dimensions
        self.client = MilvusClient(str(path), timeout=10.0)

    async def __aenter__(self):
        return self

    async def __aexit__(self, exc_type, exc_value, traceback):
        await asyncio.to_thread(self.client.close)

    def __enter__(self):
        return self

    def __exit__(self, exc_type, exc_value, traceback):
        self.client.close()

    def ready(self) -> dict:
        if not self.metadata_path.is_file():
            raise IndexNotReady("案例索引尚未完整建立，请运行重建命令")
        metadata = json.loads(self.metadata_path.read_text(encoding="utf-8"))
        if any(metadata[key] != value for key, value in self.signature.items()):
            raise IndexNotReady("Embedding 模型、地址或维度已变更，请重建索引")
        if not self.client.has_collection(COLLECTION):
            raise IndexNotReady("索引元数据存在，但案例集合缺失，请重建索引")
        return metadata

    def reset(self) -> None:
        # 仅在后端停止后重建派生索引。先撤销就绪标记，失败时不能查询半套数据。
        self.metadata_path.unlink(missing_ok=True)
        if self.client.has_collection(COLLECTION):
            self.client.drop_collection(COLLECTION)
        schema = self.client.create_schema(auto_id=False, enable_dynamic_field=False)
        schema.add_field("ticket_id", DataType.INT64, is_primary=True)
        schema.add_field(
            "content", DataType.VARCHAR, max_length=65535, enable_analyzer=True, analyzer_params={"tokenizer": "jieba"}
        )
        schema.add_field("dense_vector", DataType.FLOAT_VECTOR, dim=self.dimensions)
        schema.add_field("sparse_vector", DataType.SPARSE_FLOAT_VECTOR)
        for field in ["product_id", "module_id", "category_id"]:
            schema.add_field(field, DataType.INT64)
        schema.add_field("fingerprint", DataType.VARCHAR, max_length=64)
        schema.add_function(
            Function(
                name="content_bm25",
                function_type=FunctionType.BM25,
                input_field_names=["content"],
                output_field_names=["sparse_vector"],
            )
        )
        indexes = self.client.prepare_index_params()
        indexes.add_index("dense_vector", index_type="FLAT", metric_type="COSINE")
        indexes.add_index("sparse_vector", index_type="SPARSE_INVERTED_INDEX", metric_type="BM25")
        self.client.create_collection(COLLECTION, schema=schema, index_params=indexes)

    def add(self, rows: list[dict]) -> None:
        self.client.insert(COLLECTION, rows)

    def finish(self, count: int) -> None:
        self.client.flush(COLLECTION)
        metadata = {**self.signature, "count": count, "built_at": datetime.now(UTC).isoformat()}
        temporary = self.metadata_path.with_suffix(".tmp")
        temporary.write_text(json.dumps(metadata, ensure_ascii=False), encoding="utf-8")
        temporary.replace(self.metadata_path)

    def search(self, text: str, vector: list[float], expression: str) -> list[dict]:
        self.ready()
        # 索引由独立命令构建；新进程打开已有集合后，需要加载才能检索。
        self.client.load_collection(COLLECTION)
        requests = [
            AnnSearchRequest([vector], "dense_vector", {"metric_type": "COSINE"}, limit=20, expr=expression),
            AnnSearchRequest([text], "sparse_vector", {"metric_type": "BM25"}, limit=20, expr=expression),
        ]
        return self.client.hybrid_search(
            COLLECTION,
            reqs=requests,
            ranker=RRFRanker(60),
            limit=20,
            output_fields=["fingerprint"],
        )[0]
```

```python
import asyncio

import httpx2
from openai import APIConnectionError, APIStatusError, APITimeoutError, OpenAI
from pydantic import BaseModel, ConfigDict, FiniteFloat, StrictInt, ValidationError


class EmbeddingItem(BaseModel):
    model_config = ConfigDict(hide_input_in_errors=True)
    index: StrictInt
    embedding: list[FiniteFloat]


class EmbeddingResponse(BaseModel):
    model_config = ConfigDict(hide_input_in_errors=True)
    data: list[EmbeddingItem]


class EmbeddingError(RuntimeError):
    """模型调用失败，或返回结果不符合向量接口契约。"""


class EmbeddingClient:
    def __init__(
        self, base_url: str, api_key: str, model: str, dimensions: int, http_client: httpx2.Client | None = None
    ):
        self.model = model
        self.dimensions = dimensions
        self.client = OpenAI(base_url=base_url, api_key=api_key, timeout=30.0, max_retries=0, http_client=http_client)

    async def __aenter__(self):
        return self

    async def __aexit__(self, exc_type, exc_value, traceback):
        await asyncio.to_thread(self.client.close)

    def __enter__(self):
        return self

    def __exit__(self, exc_type, exc_value, traceback):
        self.client.close()

    def embed(self, texts: list[str]) -> list[list[float]]:
        try:
            response = self.client.embeddings.create(
                model=self.model,
                input=texts,
                dimensions=self.dimensions,
                encoding_format="float",
            )
        except APITimeoutError as error:
            raise EmbeddingError("Embedding 请求超过 30 秒，请检查模型服务") from error
        except APIConnectionError as error:
            raise EmbeddingError("无法连接 Embedding 服务，请检查地址与网络") from error
        except APIStatusError as error:
            # 上游响应正文可能包含请求内容或凭据，不透传；保留准确的 HTTP 状态。
            raise EmbeddingError(f"Embedding 服务返回 HTTP {error.status_code}") from error
        try:
            data = EmbeddingResponse.model_validate(response.model_dump()).data
        except ValidationError as error:
            raise EmbeddingError("Embedding 返回的数据结构或数值不合法") from error
        if len(data) != len(texts):
            raise EmbeddingError("Embedding 返回条数与输入不一致")
        items = sorted(data, key=lambda item: item.index)
        if [item.index for item in items] != list(range(len(texts))):
            raise EmbeddingError("Embedding 返回的 index 缺失或重复")
        vectors = [item.embedding for item in items]
        for vector in vectors:
            if len(vector) != self.dimensions:
                raise EmbeddingError("Embedding 返回的维度与配置不一致")
            if not any(vector):
                raise EmbeddingError("Embedding 返回全零向量，无法计算余弦相似度")
        return vectors
```

import asyncio

from sqlalchemy import select
from sqlalchemy.ext.asyncio import async_sessionmaker

                        .order_by(Ticket.id)

                    for start in range(0, len(tickets), 10):
                        batch = tickets[start : start + 10]
                        vectors = await asyncio.to_thread(model.embed, [case_text(ticket) for ticket in batch])
                        await asyncio.to_thread(
                            index.add, [case_row(ticket, vector) for ticket, vector in zip(batch, vectors, strict=True)]
                        )
                        print(f"已写入 {start + len(batch)} / {len(tickets)} 条有效案例", flush=True)
                await asyncio.to_thread(index.finish, len(tickets))
                print(f"索引构建完成：{len(tickets)} 条。可以重新启动后端。")

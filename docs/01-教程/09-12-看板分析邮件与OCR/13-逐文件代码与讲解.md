```python
from collections.abc import AsyncIterator, Awaitable, Callable
from contextlib import asynccontextmanager
from typing import cast
from uuid import uuid4

from fastapi import FastAPI, Request, Response
from sqlalchemy.ext.asyncio import async_sessionmaker

from app.core.database import create_database_engine
from app.core.errors import install_error_handlers
from app.core.passwords import PasswordService
from app.core.settings import Settings
from app.modules.analysis.assistance import router as assistance_router
from app.modules.analysis.evaluation import router as evaluation_router
from app.modules.analysis.router import router as analysis_router
from app.modules.config.router import router as config_router
from app.modules.dashboard.router import router as dashboard_router
from app.modules.knowledge.router import router as knowledge_router
from app.modules.knowledge.search import router as knowledge_search_router
from app.modules.tickets.import_csv import router as import_router
from app.modules.tickets.router import router as tickets_router
from app.modules.users.router import router as users_router


def create_app(settings: Settings | None = None) -> FastAPI:
    @asynccontextmanager
    async def lifespan(app: FastAPI) -> AsyncIterator[None]:
        config = settings if settings is not None else Settings.load()
        app.state.settings = config
        engine = create_database_engine(config.database_url)
        app.state.session_factory = async_sessionmaker(engine, expire_on_commit=False)
        app.state.passwords = PasswordService()
        try:
            yield
        finally:
            await engine.dispose()

    app = FastAPI(title="售后工单智能辅助系统", lifespan=lifespan)

    @app.middleware("http")
    async def request_id(request: Request, call_next: Callable[[Request], Awaitable[Response]]) -> Response:
        request.state.request_id = uuid4().hex
        response = await call_next(request)
        response.headers["X-Request-ID"] = cast(str, request.state.request_id)
        return response

    install_error_handlers(app)
    app.include_router(users_router)
    app.include_router(config_router)
    app.include_router(import_router)
    app.include_router(tickets_router)
    app.include_router(analysis_router)
    app.include_router(assistance_router)
    app.include_router(evaluation_router)
    app.include_router(dashboard_router)
    app.include_router(knowledge_router)
    app.include_router(knowledge_search_router)

    @app.get("/api/health", tags=["健康检查"])
    async def health() -> dict[str, str]:
        return {"status": "ok"}

    return app


app = create_app()
```

```python
from typing import Annotated

from fastapi import APIRouter, Query
from pydantic import BaseModel
from sqlalchemy import extract, func, select

from app.core.security import CurrentUser, DbSession
from app.models import Ticket

router = APIRouter(prefix="/api/dashboard", tags=["统计看板"])


class DashboardView(BaseModel):
    total: int
    by_status: dict[str, int]
    by_priority: dict[str, int]
    resolution_samples: int
    average_resolution_hours: float | None


@router.get("", response_model=DashboardView)
async def dashboard(db: DbSession, user: CurrentUser, product_id: Annotated[int | None, Query(gt=0)] = None):
    filters = [] if product_id is None else [Ticket.product_id == product_id]
    # 同一条聚合查询返回所有指标，避免并发修改时各卡片使用不同数据快照。
    statuses = ["pending", "processing", "resolved", "closed"]
    priorities = ["P1", "P2", "P3", "P4"]
    resolved = Ticket.status.in_(["resolved", "closed"]) & Ticket.resolved_at.is_not(None)
    columns = [func.count().label("total")]
    columns += [func.count().filter(Ticket.status == key).label(key) for key in statuses]
    columns += [func.count().filter(Ticket.priority == key).label(key) for key in priorities]
    columns += [
        func.count().filter(resolved).label("samples"),
        func.avg(extract("epoch", Ticket.resolved_at - Ticket.created_at) / 3600).filter(resolved).label("average"),
    ]
    result = (await db.execute(select(*columns).select_from(Ticket).where(*filters))).mappings().one()
    return DashboardView(
        total=result["total"],
        by_status={key: result[key] for key in statuses},
        by_priority={key: result[key] for key in priorities},
        resolution_samples=result["samples"],
        average_resolution_hours=None if result["average"] is None else round(float(result["average"]), 2),
    )
```

```python
from datetime import UTC, datetime
from typing import Annotated

from fastapi import APIRouter, HTTPException, Query
from sqlalchemy import func, or_, select

from app.core.security import CurrentUser, DbSession
from app.models import Ticket
from app.modules.tickets.schemas import (
    CloseRequest,
    NoteRequest,
    Priority,
    ResolveRequest,
    TicketCreate,
    TicketDetail,
    TicketPage,
    TicketPatch,
    TicketStatus,
)
from app.modules.tickets.service import (
    add_record,
    get_ticket,
    lock_ticket,
    validate_catalog,
    validate_owner,
)

router = APIRouter(prefix="/api/tickets", tags=["工单"])


@router.post("", response_model=TicketDetail, status_code=201)
async def create_ticket(body: TicketCreate, db: DbSession, user: CurrentUser):
    await validate_catalog(db, body.product_id, body.module_id, body.category_id)
    # model_dump 把校验后的请求转成字典，** 将字段作为参数交给 Ticket。
    ticket = Ticket(**body.model_dump(), owner_id=user.id)
    add_record(ticket, user.id, "created", "创建工单，状态为待处理")
    db.add(ticket)
    # 工单和处理记录一起提交，避免只保存其中一项。
    await db.commit()
    return ticket


@router.get("", response_model=TicketPage)
async def list_tickets(
    db: DbSession,
    user: CurrentUser,
    page: Annotated[int, Query(ge=1)] = 1,
    page_size: Annotated[int, Query(ge=1, le=100)] = 20,
    keyword: Annotated[str | None, Query(max_length=200)] = None,
    status: TicketStatus | None = None,
    priority: Priority | None = None,
    product_id: Annotated[int | None, Query(gt=0)] = None,
    module_id: Annotated[int | None, Query(gt=0)] = None,
    category_id: Annotated[int | None, Query(gt=0)] = None,
    owner_id: Annotated[int | None, Query(gt=0)] = None,
):
    filters = []
    if keyword:
        filters.append(or_(Ticket.title.ilike(f"%{keyword}%"), Ticket.description.ilike(f"%{keyword}%")))
    for column, value in [
        (Ticket.status, status),
        (Ticket.priority, priority),
        (Ticket.product_id, product_id),
        (Ticket.module_id, module_id),
        (Ticket.category_id, category_id),
        (Ticket.owner_id, owner_id),
    ]:
        if value is not None:
            filters.append(column == value)
    total = await db.scalar(select(func.count()).select_from(Ticket).where(*filters))
    # 查询条件与总数保持一致；offset 跳过前面各页，limit 限制本页条数。
    query = (
        select(Ticket)
        .where(*filters)
        .order_by(Ticket.id.desc())
        .offset((page - 1) * page_size)
        .limit(page_size)
    )
    result = await db.scalars(query)
    items = result.all()
    return {"items": items, "total": total, "page": page, "page_size": page_size}


@router.get("/{ticket_id}", response_model=TicketDetail)
async def ticket_detail(ticket_id: int, db: DbSession, user: CurrentUser):
    return await get_ticket(db, ticket_id)


@router.patch("/{ticket_id}", response_model=TicketDetail)
async def update_ticket(ticket_id: int, body: TicketPatch, db: DbSession, user: CurrentUser):
    ticket = await lock_ticket(db, ticket_id)
    # PATCH 只修改请求里实际出现的字段，省略的字段保留原值。
    changes = body.model_dump(exclude_unset=True)
    await validate_catalog(
        db,
        changes.get("product_id", ticket.product_id),
        changes.get("module_id", ticket.module_id),
        changes.get("category_id", ticket.category_id),
        current=ticket,
    )
    if "owner_id" in changes and changes["owner_id"] != ticket.owner_id:
        await validate_owner(db, changes["owner_id"])
    for field, value in changes.items():
        setattr(ticket, field, value)
    add_record(ticket, user.id, "updated", "更新了工单信息")
    await db.commit()
    return ticket


@router.post("/{ticket_id}/records", response_model=TicketDetail, status_code=201)
async def write_note(ticket_id: int, body: NoteRequest, db: DbSession, user: CurrentUser):

@router.post("/{ticket_id}/start", response_model=TicketDetail)
async def start_ticket(ticket_id: int, db: DbSession, user: CurrentUser):
    ticket = await lock_ticket(db, ticket_id)
    if ticket.status != "pending":
        raise HTTPException(status_code=409, detail="只有待处理工单可以开始处理")
    ticket.status = "processing"
    add_record(ticket, user.id, "status", "开始处理：待处理 → 处理中")
    await db.commit()
    return ticket


@router.post("/{ticket_id}/resolve", response_model=TicketDetail)
async def resolve_ticket(ticket_id: int, body: ResolveRequest, db: DbSession, user: CurrentUser):
    ticket = await lock_ticket(db, ticket_id)
    if ticket.status != "processing":
        raise HTTPException(status_code=409, detail="只有处理中工单可以标记已解决")
    ticket.solution = body.solution
    ticket.status = "resolved"
    ticket.resolved_at = datetime.now(UTC)
    add_record(ticket, user.id, "status", "保存解决方案：处理中 → 已解决")
    await db.commit()
    return ticket


@router.post("/{ticket_id}/close", response_model=TicketDetail)
async def close_ticket(ticket_id: int, body: CloseRequest, db: DbSession, user: CurrentUser):
    ticket = await lock_ticket(db, ticket_id)
    if ticket.status != "resolved":
        raise HTTPException(status_code=409, detail="只有已解决工单可以关闭")
    ticket.status = "closed"
    add_record(ticket, user.id, "status", "客服确认已完成客户回复：已解决 → 已关闭")
    await db.commit()
    return ticket
```

```python
"""记录真实解决时刻，历史缺失时间不编造。"""

import sqlalchemy as sa
from alembic import op

revision = "09_resolution_time"
down_revision = "07_assistance"
branch_labels = None
depends_on = None


def upgrade():
    op.add_column("tickets", sa.Column("resolved_at", sa.DateTime(timezone=True), nullable=True))
    op.execute("""UPDATE tickets t SET resolved_at = r.solved_at FROM
        (SELECT ticket_id, min(created_at) AS solved_at FROM ticket_records
         WHERE kind='status' AND content='保存解决方案：处理中 → 已解决'
         GROUP BY ticket_id) r
        WHERE t.id=r.ticket_id AND t.status IN ('resolved','closed')
          AND r.solved_at >= t.created_at""")


def downgrade():
    op.drop_column("tickets", "resolved_at")
```

### frontend/src/router/index.ts

`frontend/src/router/index.ts`（源码见本教程对应代码块）

**它做什么：** 为统计页面登记按需加载路由，其余登录和管理员守卫继续沿用。

**固定写法：** createRouter、动态 import 是框架和语言接口。

**能改什么：** 路由地址可以成套修改，不能删权限守卫。

<!-- source: 09-dashboard/frontend/src/router/index.ts -->
```typescript
import { createRouter, createWebHistory } from 'vue-router'
import { useAuthStore } from '../stores/auth'

export const router = createRouter({
  history: createWebHistory(),
  routes: [
    { path: '/', redirect: '/tickets' },
    { path: '/dashboard', component: () => import('../views/DashboardView.vue') },
    { path: '/tickets', component: () => import('../views/TicketsView.vue') },
    { path: '/tickets/import', component: () => import('../views/ImportView.vue'), meta: { admin: true } },
    { path: '/tickets/new', component: () => import('../views/TicketEditor.vue') },
    { path: '/tickets/:id(\\d+)/edit', component: () => import('../views/TicketEditor.vue') },
    { path: '/tickets/:id(\\d+)', component: () => import('../views/TicketDetailView.vue') },
    { path: '/knowledge', component: () => import('../views/KnowledgeView.vue') },
    { path: '/knowledge/new', component: () => import('../views/KnowledgeEditor.vue'), meta: { admin: true } },
    { path: '/knowledge/:id(\\d+)', component: () => import('../views/KnowledgeEditor.vue') },
    { path: '/health', component: () => import('../views/HealthView.vue'), meta: { public: true } },
    { path: '/login', component: () => import('../views/LoginView.vue'), meta: { public: true, standalone: true } },
    { path: '/admin', component: () => import('../views/AdminView.vue'), meta: { admin: true } },
    { path: '/:pathMatch(.*)*', component: () => import('../views/NotFoundView.vue'), meta: { public: true } },
  ],
})

router.beforeEach(async to => {
  const auth = useAuthStore()
  if (auth.token && !auth.user && to.path !== '/login') {
    try {
      await auth.restoreSession()
    } catch (error) {
      if (!(error instanceof Error)) throw error
      // 公开页面和连接检查不依赖登录，后端故障时仍允许访问。
      if (to.meta.public) return true
      return { path: '/login', query: { redirect: to.fullPath, session_error: `恢复登录失败：${error.message}` } }
    }
  }
  if (!to.meta.public && !auth.token) return { path: '/login', query: { redirect: to.fullPath } }
  if (to.meta.admin && auth.user?.role !== 'admin') return '/tickets'
})
```

```vue
  if (busy.value) return

  errorMessage.value = ''
  metrics.value = null

  } catch (error) {
    if (!(error instanceof Error)) throw error
    errorMessage.value = error.message
  } finally {
    busy.value = false
  }
}

onMounted(async () => {

    products.value = (await getCatalog(auth.token)).products

  } catch (error) {
    if (!(error instanceof Error)) throw error
    errorMessage.value = error.message
  }
})
```

```vue
    const result = await getAnalysis(auth.token,props.ticketId)
```

```python
import smtplib
import ssl
from email.message import EmailMessage
from email.utils import format_datetime
from datetime import datetime, timezone
from pathlib import Path
from typing import Literal

from pydantic import EmailStr, Field, SecretStr, TypeAdapter, field_validator
from pydantic_settings import BaseSettings, SettingsConfigDict


class SmtpSettings(BaseSettings):
    model_config = SettingsConfigDict(
        env_file=Path(__file__).resolve().parents[2] / ".env.smtp",
        env_file_encoding="utf-8",
        hide_input_in_errors=True,
    )
    smtp_host: str = Field(min_length=1)
    smtp_port: int = Field(ge=1, le=65535)
    smtp_security: Literal["ssl", "starttls"]
    smtp_username: str = Field(min_length=1)
    smtp_password: SecretStr = Field(min_length=1)
    smtp_from: EmailStr
    smtp_test_recipients: str = Field(min_length=1)

    @field_validator("smtp_test_recipients")
    @classmethod
    def valid_recipients(cls, value):
        TypeAdapter(list[EmailStr]).validate_python([part.strip() for part in value.split(",")])
        return value

    def allowed(self, recipient: str) -> bool:
        return recipient.casefold() in {
            part.strip().casefold() for part in self.smtp_test_recipients.split(",")
        }


class DeliveryRejected(RuntimeError):
    """SMTP 明确拒绝，本次未被接受。"""


class DeliveryUnknown(RuntimeError):
    """发送结果无法确定，不得自动重发。"""


def deliver(config: SmtpSettings, body: dict) -> None:
    message = EmailMessage()
    message["From"] = str(config.smtp_from)
    message["To"] = body["recipient"]
    message["Subject"] = body["subject"]
    message["Message-ID"] = body["message_id"]
    message["Date"] = format_datetime(datetime.now(timezone.utc))
    message.set_content(body["content"])
    client = None
    try:
        context = ssl.create_default_context()
        if config.smtp_security == "ssl":
            client = smtplib.SMTP_SSL(
                config.smtp_host, config.smtp_port, timeout=15, context=context
            )
        else:
            client = smtplib.SMTP(config.smtp_host, config.smtp_port, timeout=15)
            client.starttls(context=context)
        client.login(config.smtp_username, config.smtp_password.get_secret_value())
        client.send_message(message, from_addr=str(config.smtp_from), to_addrs=[body["recipient"]])
    except smtplib.SMTPAuthenticationError as error:
        raise DeliveryRejected("SMTP 认证失败，请检查授权码和账号") from error
    except smtplib.SMTPRecipientsRefused as error:
        raise DeliveryRejected("SMTP 拒绝收件人") from error
    except (smtplib.SMTPSenderRefused, smtplib.SMTPDataError) as error:
        raise DeliveryRejected(f"SMTP 明确拒绝发送，代码 {error.smtp_code}") from error
    except smtplib.SMTPNotSupportedError as error:
        raise DeliveryRejected("SMTP 服务不支持所配置的安全连接或认证方式") from error
    except (OSError, smtplib.SMTPException) as error:
        # 连接中断可能发生在服务器已经接受邮件之后，不能当作可重试失败。
        raise DeliveryUnknown(
            "邮件连接异常，无法确认 SMTP 是否接受；请核查邮箱和服务端记录"
        ) from error
    finally:
        if client is not None:
            client.close()
```

```python
    if existing is not None:
        return existing
    await lock_ticket(db, ticket_id)

        return existing

        subject=body.subject,
        content=body.content,
        message_id=f"<{body.request_id}@{str(config.smtp_from).split('@')[1]}>",
        status="sending",
    )
    db.add(row)
    await db.commit()
    # 先持久化人工确认的内容，再进行不可回滚的外部投递。
    try:
        (await asyncio.to_thread(smtp.deliver,
            config,
            {
                "recipient": row.recipient,
                "subject": row.subject,
                "content": row.content,
                "message_id": row.message_id,
            },
        ))
        row.status = "accepted"
    except smtp.DeliveryRejected as error:
        row.status = "failed"
        row.error = str(error)
    except smtp.DeliveryUnknown as error:
        row.status = "unknown"
        row.error = str(error)
    row.finished_at = datetime.now(timezone.utc)
    await db.commit()
    return row
```

```vue
    errorMessage.value = error.message
  } finally {
    loading.value = false
  }

  try {
    ticket.value = await getTicket(auth.token, ticketId)
    names.value.category = categories.value[ticket.value.category_id] ?? ''

  catch (error) {
    if (!(error instanceof Error)) throw error
    errorMessage.value = error.message
  }
}

// 开始处理和添加记录都会替换详情；409 时先刷新，让按钮随真实状态变化。
async function perform(action: 'start' | 'note') {
  if (action === 'note' && !note.value.trim()) return
  busy.value = true
  errorMessage.value = ''
  try {
    if (action === 'start') {
      ticket.value = await startTicket(auth.token, ticketId)
    } else {
      ticket.value = await addNote(auth.token, ticketId, note.value.trim())
      note.value = ''
    }
  } catch (error) {
    if (!(error instanceof Error)) throw error
    if (error instanceof ApiError && error.status === 409) await refreshTicket()
    errorMessage.value = error.message
  } finally {
    busy.value = false
  }
}

function start() {
  void perform('start')
}

function openResolve() {

  resolveError.value = ''

async function confirmResolve() {

    resolveError.value = '请填写有效的解决方案'

  resolveError.value = ''

    ticket.value = await resolveTicket(auth.token, ticketId, resolveText.value.trim())

    if (error instanceof ApiError && error.status === 409) {
      await refreshTicket()
      resolveOpen.value = false
      errorMessage.value = error.message
    } else {
      resolveError.value = error.message
    }
  } finally {
    busy.value = false
  }
}

function openClose() {
  replyConfirmed.value = false
  closeError.value = ''
  closeOpen.value = true
}

async function confirmClose() {

    ticket.value = await closeTicket(auth.token, ticketId)
    closeOpen.value = false

    if (!(error instanceof Error)) throw error

      await refreshTicket()
      closeOpen.value = false
      errorMessage.value = error.message
    } else {
      closeError.value = error.message
    }
  } finally {
    busy.value = false
  }
}
```

```python
import json
import subprocess
import sys
from pathlib import Path
from tempfile import TemporaryDirectory
from threading import BoundedSemaphore

from app.integrations.ocr_worker import ExtractionError

slot = BoundedSemaphore(1)


def extract_file(path: Path) -> str:
    if not slot.acquire(blocking=False):
        raise ExtractionError("当前有资料正在识别，请完成后再上传")
    try:
        with TemporaryDirectory() as temp:
            output = Path(temp) / "result.json"
            try:
                completed = subprocess.run(
                    [sys.executable, "-m", "app.integrations.ocr_worker", str(path), str(output)],
                    cwd=Path(__file__).resolve().parents[2],
                    stdout=subprocess.DEVNULL,
                    stderr=subprocess.DEVNULL,
                    timeout=180,
                    check=False,
                )
            except subprocess.TimeoutExpired as error:
                raise ExtractionError(
                    "资料识别超过 180 秒，进程已终止；请拆分文件后再试"
                ) from error
            if completed.returncode != 0 or not output.exists():
                raise ExtractionError("资料解析进程失败，请检查已安装的 PDF/OCR 依赖")
            result = json.loads(output.read_text(encoding="utf-8"))
            if "error" in result:
                raise ExtractionError(result["error"])
            return result["content"]
    finally:
        slot.release()
```

### backend/app/integrations/ocr_worker.py

`backend/app/integrations/ocr_worker.py`（源码见本教程对应代码块）

**它做什么：** 在独立进程验证图片与 PDF，直接读取文本或识别扫描页，再返回正文或明确错误。

**固定写法：** PaddleOCR.predict、rec_texts、pymupdf.open、get_pixmap 是库接口。

`page.get_text("text")` 固定要求文字结果，紧跟的 `assert isinstance(page_text, str)` 把这一点写清楚，方便类型检查；它不负责把错误内容变成空正文。`enable_mkldnn=False` 是针对本机 PaddlePaddle 3.3.1 推理报错的实测处理。

**能改什么：** 限制和模型可经实测调整；模型参数与版本、文档提示、测试要一起改。

<!-- source: 12-ocr/backend/app/integrations/ocr_worker.py -->
```python
"""独立进程内解析 PDF/图片，入口只接收本机临时文件。"""

import json
import sys
from contextlib import redirect_stderr, redirect_stdout
from pathlib import Path
from tempfile import TemporaryDirectory
from functools import lru_cache


class ExtractionError(RuntimeError):
    pass


@lru_cache(maxsize=1)
def get_ocr():
    from pydantic import ValidationError
    from app.core.ocr_settings import OcrSettings

    try:
        config = OcrSettings.model_validate({})
    except ValidationError as error:
        raise ExtractionError(
            "OCR 本地模型配置无效，请检查 backend/.env.ocr 中的模型目录"
        ) from error
    try:
        from paddleocr import PaddleOCR

        return PaddleOCR(
            device="cpu",
            # 当前 PaddlePaddle 3.3.1 的 oneDNN 推理会报 PIR 属性转换错误。
            enable_mkldnn=False,
            text_detection_model_name="PP-OCRv5_mobile_det",
            text_recognition_model_name="PP-OCRv5_mobile_rec",
            text_detection_model_dir=str(config.ocr_detection_model_dir),
            text_recognition_model_dir=str(config.ocr_recognition_model_dir),
            use_doc_orientation_classify=False,
            use_doc_unwarping=False,
            use_textline_orientation=False,
            cpu_threads=2,
        )
    except (ImportError, RuntimeError, ValueError, OSError) as error:
        raise ExtractionError("OCR 引擎初始化失败，请检查 CPU 依赖与本地模型文件") from error


def recognize(path: Path) -> str:
    try:
        results = get_ocr().predict(str(path))
        return "\n".join(text for result in results for text in result["rec_texts"]).strip()
    except (RuntimeError, ValueError, OSError) as error:
        if isinstance(error, ExtractionError):
            raise
        raise ExtractionError("OCR 识别失败，请检查模型文件及输入图片") from error


def extract_content(path: Path) -> str:
    import pymupdf
    from PIL import Image, UnidentifiedImageError

    if path.suffix.lower() in {".jpg", ".jpeg", ".png"}:
        try:
            with Image.open(path) as image:
                if image.format not in {"JPEG", "PNG"}:
                    raise ExtractionError("文件内容不是 JPG 或 PNG")
                if image.width * image.height > 16000000:
                    raise ExtractionError("图片不能超过 1600 万像素")
                image.verify()
        except (UnidentifiedImageError, OSError, Image.DecompressionBombError) as error:
            raise ExtractionError("图片损坏、格式不正确或尺寸过大") from error
        content = recognize(path)
    else:
        texts = []
        try:
            with pymupdf.open(path) as document, TemporaryDirectory(dir=path.parent) as temp:
                if not document.is_pdf or document.needs_pass:
                    raise ExtractionError("仅支持未加密的 PDF")
                if not 1 <= len(document) <= 20:
                    raise ExtractionError("PDF 必须为 1—20 页")
                for index, page in enumerate(document):
                    page_text = page.get_text("text", sort=True)
                    assert isinstance(page_text, str)
                    text = page_text.strip()
                    if not text:
                        rect = page.rect
                        if not 0 < rect.width <= 2000 or not 0 < rect.height <= 2000:
                            raise ExtractionError("PDF 页面尺寸超出支持范围")
                        scale = min(150 / 72, 4000 / max(rect.width, rect.height))
                        image_path = Path(temp) / f"page-{index + 1}.png"
                        page.get_pixmap(matrix=pymupdf.Matrix(scale, scale), alpha=False).save(
                            image_path
                        )
                        text = recognize(image_path)
                    if text:
                        texts.append(f"第{index + 1}页\n{text}")
                    if sum(len(item) for item in texts) > 100000:
                        raise ExtractionError("提取正文不能超过 100000 字")
        except (pymupdf.FileDataError, pymupdf.EmptyFileError) as error:
            raise ExtractionError("PDF 损坏或内容为空") from error
        content = "\n\n".join(texts)
    if not content.strip():
        raise ExtractionError("未提取到有效文字，请换用清晰资料或手工录入")
    if len(content) > 100000:
        raise ExtractionError("提取正文不能超过 100000 字")
    return content


def main():
    source, output = map(Path, sys.argv[1:3])
    try:
        # 原生库输出留在子进程中，不混入协议，也不打印文档正文。
        with open("/dev/null", "w") as quiet, redirect_stdout(quiet), redirect_stderr(quiet):
            content = extract_content(source)
        result = {"content": content}
    except ExtractionError as error:
        result = {"error": str(error)}
    output.write_text(json.dumps(result, ensure_ascii=False), encoding="utf-8")


if __name__ == "__main__":
    main()
```

```python
@router.post('/upload', response_model=DocumentView, status_code=201)
async def upload_document(file: UploadFile, title: Annotated[str, Form()],
                    product_id: Annotated[int, Form(gt=0)], module_id: Annotated[int, Form(gt=0)],
                    category_id: Annotated[int, Form(gt=0)], db: DbSession, user: AdminUser):
    filename = file.filename or ''
    suffix = Path(filename).suffix.lower()
    if suffix not in {'.md', '.txt', '.pdf', '.jpg', '.jpeg', '.png'}:
        raise HTTPException(422, '只支持 MD、TXT、PDF、JPG 和 PNG 文件')
    await validate_catalog(db, product_id, module_id, category_id)
    limit = 256 * 1024 if suffix in {'.md', '.txt'} else 10 * 1024 * 1024
    try:
        raw = await file.read(limit + 1)
    finally:
        await file.close()
    if len(raw) > limit:
        raise HTTPException(413, '文本不能超过 256 KiB，PDF 和图片不能超过 10 MiB')
    if suffix in {'.md', '.txt'}:
        try:
            content = raw.decode('utf-8-sig')
        except UnicodeDecodeError as error:
            raise HTTPException(422, '文本文档必须使用 UTF-8 编码') from error
    else:
        with TemporaryDirectory() as temp:
            source = Path(temp) / ('source' + suffix)
            source.write_bytes(raw)
            try:
                content = await asyncio.to_thread(document_extract.extract_file, source)
            except document_extract.ExtractionError as error:
                raise HTTPException(422, str(error)) from error
    # 上传和直接录入使用同一套边界规则；错误不包含正文。
    from pydantic import ValidationError
    try:
        body = DocumentInput(title=title, content=content, product_id=product_id,
                             module_id=module_id, category_id=category_id)
    except ValidationError as error:
        raise HTTPException(422, '标题须为1—200字，正文须为1—100000字') from error
    document = KnowledgeDocument(**body.model_dump(), source_name=filename.replace('\\', '/').split('/')[-1][:200])
    db.add(document)
    await db.commit()
    return document
```

```typescript
// 路由切换与关闭标签都可能丢失内存草稿，集中处理这两个离开入口。
export function useUnsavedChanges(hasChanges: () => boolean) {
  onBeforeRouteLeave(() => !hasChanges() || window.confirm('当前编辑或操作尚未完成，确定离开吗？'))

  function beforeUnload(event: BeforeUnloadEvent) {
    if (!hasChanges()) return
    event.preventDefault()
    event.returnValue = ''
  }

  onMounted(() => window.addEventListener('beforeunload', beforeUnload))
  onBeforeUnmount(() => window.removeEventListener('beforeunload', beforeUnload))
}
```

## 异步资源释放回归检查

### backend/tests/test_async_lifecycle.py

启动恢复或管理员初始化遇到数据库错误时，错误应继续暴露，同时释放异步连接池。这两项测试只替换故障点，不连接模型。

<!-- source: 12-ocr/backend/tests/test_async_lifecycle.py -->
```python
import asyncio
from contextlib import asynccontextmanager
from types import SimpleNamespace
from unittest.mock import AsyncMock

import pytest

from app import main, seed


def test_startup_failure_releases_database_engine(monkeypatch):
    engine = SimpleNamespace(dispose=AsyncMock())
    monkeypatch.setattr(main, "engine", engine)
    monkeypatch.setattr(main, "interrupt_analyses", AsyncMock(side_effect=RuntimeError("恢复失败")))

    async def start():
        with pytest.raises(RuntimeError, match="恢复失败"):
            async with main.lifespan(main.app):
                pytest.fail("启动失败后不应开始接收请求")

    asyncio.run(start())
    engine.dispose.assert_awaited_once()


def test_seed_database_failure_releases_engine(monkeypatch):
    @asynccontextmanager
    async def failed_connection():
        raise RuntimeError("数据库不可用")
        yield  # 声明异步上下文管理器；连接失败时不会到这里。

    engine = SimpleNamespace(dispose=AsyncMock())
    monkeypatch.setattr(seed, "engine", engine, raising=False)
    monkeypatch.setattr(seed, "SessionFactory", SimpleNamespace(begin=failed_connection))
    monkeypatch.setattr("sys.argv", ["seed", "--username", "admin", "--display-name", "管理员"])
    monkeypatch.setattr(seed, "getpass", lambda prompt: "Only-for-test-8392")

    with pytest.raises(RuntimeError, match="数据库不可用"):
        asyncio.run(seed.main())
    engine.dispose.assert_awaited_once()
```
